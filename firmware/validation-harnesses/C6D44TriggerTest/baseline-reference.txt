void queuePartial(){if(cur&&cur->count){xQueueSend(readyQ,&cur,portMAX_DELAY);cur=nullptr;}}
void ignoreSample(uint16_t v){++ignored;++silenceCount;silenceSum+=v;if(v<silenceMin)silenceMin=v;if(v>silenceMax)silenceMax=v;}
void ringPush(uint16_t v){if(ringCount==PRE_SAMPLES){ignoreSample(preRing[ringHead]);}else ++ringCount;preRing[ringHead]=v;if(++ringHead==PRE_SAMPLES)ringHead=0;}

void beginEvent(uint64_t trigger){Event&e=events[nextEvent];e.id=nextEvent+1;e.trigger=trigger;e.pre=ringCount;e.start=trigger-ringCount;e.activity=EVENT_SAMPLES;e.post=POST_SAMPLES;e.activityEnd=trigger+EVENT_SAMPLES;e.end=e.activityEnd+POST_SAMPLES;lastQueuedIndex=UINT64_MAX;uint32_t pos=(ringHead+PRE_SAMPLES-ringCount)%PRE_SAMPLES;for(uint32_t i=0;i<ringCount;i++){appendRaw(preRing[pos],e.start+i,nextEvent);if(++pos==PRE_SAMPLES)pos=0;}ringCount=0;currentEvent=nextEvent++;}
void endEvent(){queuePartial();currentEvent=-1;lastQueuedIndex=UINT64_MAX;}

void writerTask(void*){while(true){Buffer*b=nullptr;if(xQueueReceive(readyQ,&b,pdMS_TO_TICKS(20))==pdTRUE){Event&e=events[b->event];size_t req=b->count*2;if(e.rawOffset==UINT64_MAX)e.rawOffset=raw.position();uint64_t wb=esp_timer_get_time();size_t n=raw?raw.write((uint8_t*)b->samples,req):0;uint32_t dt=uint32_t(esp_timer_get_time()-wb);++e.writes;e.writeUs+=dt;if(dt>e.writeMaxUs)e.writeMaxUs=dt;size_t ns=n/2;e.rawStored+=ns;e.rawBytes+=n;e.crc=esp_crc32_le(e.crc,(uint8_t*)b->samples,n);rawStored+=ns;rawBytes+=n;if(n!=req){++shortWrites;++sdErrors;++e.shortWrites;++e.sdErrors;if(n==0){++zeroWrites;++e.zeroWrites;}uint64_t missing=b->count-ns;lost+=missing;e.lost+=missing;Serial.printf("{\"type\":\"SD_ERROR\",\"event\":%llu,\"requested\":%u,\"actual\":%u}\n",e.id,unsigned(req),unsigned(n));}b->count=0;xQueueSend(freeQ,&b,portMAX_DELAY);}if(stopping&&uxQueueMessagesWaiting(readyQ)==0){if(raw)raw.flush();stopping=false;}}}
